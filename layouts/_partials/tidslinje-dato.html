January 2006" }}{{ end -}} {{- else -}} {{- with findRESubmatch `^(\d{1,2})\.\s+(\p{L}+)\.?\s+(\d{4})$` $r 1 -}} {{- $s := index . 0 -}} {{- with index $maaneder (lower (index $s 2)) -}} {{- $kandidat := printf "%s-%02d-%02d" (index $s 3) .
